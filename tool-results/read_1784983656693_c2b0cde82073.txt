   290→  const [timeoutSeconds, setTimeoutSeconds] = React.useState(0);
   291→  React.useEffect(() => setMounted(true), []);
   292→
   293→  // Initialize locale from localStorage
   294→  const setLocale = useApp((s) => s.setLocale);

   296→    const saved = localStorage.getItem("tp_locale");

   298→  }, [setLocale]);
   299→
   300→  const isAdmin = user?.role === "ADMIN";
   301→  const navGroups: NavGroup[] = isAdmin ? ADMIN_NAV : USER_NAV;
   302→
   303→  const doLogout = React.useCallback(async (reason?: string) => {
   304→    try {
   305→      await apiPost("/api/auth/logout", {});
   306→    } catch {
   307→      /* ignore */
   308→    }
   309→    logoutClient();
   310→    toast.success(reason ?? "Signed out");
   311→  }, [logoutClient]);
   312→
   313→  // ─── Inactivity timeout — auto-logout after 15 minutes with 2-min warning ───
   314→  const INACTIVITY_MS = 15 * 60 * 1000; // 15 minutes total
   315→  const WARNING_MS = 13 * 60 * 1000; // show warning at 13 minutes (2 min before logout)
   316→  const WARNING_DURATION_S = 120; // 2 minute countdown
   317→  React.useEffect(() => {
   318→    let logoutTimer: ReturnType<typeof setTimeout>;
   319→    let warningTimer: ReturnType<typeof setTimeout>;
   320→    let countdownInterval: ReturnType<typeof setInterval>;
   321→
   322→    const clearAll = () => {
   323→      clearTimeout(logoutTimer);
   324→      clearTimeout(warningTimer);
   325→      clearInterval(countdownInterval);
   326→    };
   327→
   328→    const resetTimer = () => {
   329→      clearAll();
   330→      setShowTimeoutWarning(false);
   331→
   332→      // Warning timer — fires at 13 minutes
   333→      warningTimer = setTimeout(() => {
   334→        setShowTimeoutWarning(true);
   335→        setTimeoutSeconds(WARNING_DURATION_S);
   336→
   337→        // Countdown timer — updates every second
   338→        let remaining = WARNING_DURATION_S;
   339→        countdownInterval = setInterval(() => {
   340→          remaining -= 1;
   341→          setTimeoutSeconds(remaining);
   342→          if (remaining <= 0) {
   343→            clearInterval(countdownInterval);
   344→          }
   345→        }, 1000);
   346→
   347→        // Logout timer — fires at 15 minutes
   348→        logoutTimer = setTimeout(() => {
   349→          doLogout("You've been signed out due to inactivity");
   350→        }, INACTIVITY_MS - WARNING_MS);
   351→      }, WARNING_MS);
   352→    };
   353→
   354→    // Reset on user activity
   355→    const events = ["mousemove", "keydown", "click", "scroll", "touchstart"];
   356→    events.forEach((e) => window.addEventListener(e, resetTimer, { passive: true }));
   357→    resetTimer();
   358→
   359→    return () => {
   360→      clearAll();
   361→      events.forEach((e) => window.removeEventListener(e, resetTimer));
   362→    };
   363→  }, [doLogout]);
   364→

   739→/** Notifications dialog — shows the user's in-app notifications. */
   740→function NotificationsDialog({ open, onOpenChange }: { open: boolean; onOpenChange: (o: boolean) => void }) {
   741→  const { data: resp, isLoading } = useApi<{ items: NotificationItem[]; unreadCount: number } | null>(open ? "/api/notifications" : null);
   742→  const notifications = resp?.items ?? [];
   743→  const [expandedId, setExpandedId] = React.useState<string | null>(null);
   744→
   745→  const markAllRead = async () => {
   746→    if (!notifications) return;
   747→    try {
   748→      await Promise.all(
   749→        notifications.filter((n) => !n.read).map((n) =>
   750→          apiPost(`/api/notifications/${n.id}/read`, {}).catch(() => null)
   751→        )
   752→      );
   753→      mutateApi("/api/notifications");
   754→      toast.success("All notifications marked as read");
   755→    } catch (e: any) {
   756→      if (e?.status === 401) return;
   757→      toast.error("Could not mark notifications as read");
   758→    }
   759→  };
   760→
   761→  const markOneRead = async (id: string) => {
   762→    try {
   763→      await apiPost(`/api/notifications/${id}/read`, {});
   764→      mutateApi("/api/notifications");
   765→    } catch (e: any) {
   766→      if (e?.status === 401) return;
   767→    }
   768→  };
   769→
   770→  return (
   771→    <>
   772→      {/* Backdrop — click anywhere to close */}
   773→      {open && <div className="fixed inset-0 z-[55]" onClick={() => onOpenChange(false)} />}
   774→
   775→      {/* Notification panel — pinned top-right, not centered */}
   776→      {open && (
   777→        <div className="fixed right-3 top-16 z-[56] w-[calc(100vw-1.5rem)] max-w-sm rounded-xl border bg-popover shadow-lg sm:right-4 sm:top-16">
   778→          {/* Header */}
   779→          <div className="flex items-center justify-between border-b px-4 py-3">
   780→            <div className="flex items-center gap-2">
   781→              <Bell className="h-4 w-4 text-primary" />
   782→              <span className="text-sm font-semibold">Notifications</span>
   783→              {resp?.unreadCount ? (
   784→                <span className="rounded-full bg-primary px-1.5 py-0.5 text-[10px] font-medium text-primary-foreground">{resp.unreadCount}</span>
   785→              ) : null}
   786→            </div>
   787→            <div className="flex items-center gap-2">
   788→              {notifications.some((n) => !n.read) && (
   789→                <button onClick={markAllRead} className="text-[11px] text-muted-foreground hover:text-primary">Mark all read</button>
   790→              )}
   791→              <button onClick={() => onOpenChange(false)} className="text-muted-foreground hover:text-foreground">
   792→                <X className="h-4 w-4" />
   793→              </button>
   794→            </div>
   795→          </div>
   796→
   797→          {/* List — scrollable, max height so it doesn't cover the screen */}
   798→          <div className="max-h-[60vh] overflow-y-auto p-2">
   799→            {isLoading ? (
   800→              <div className="py-8 text-center text-sm text-muted-foreground">Loading…</div>
   801→            ) : notifications.length > 0 ? (
   802→              <div className="space-y-1">
   803→                {notifications.map((n) => {
   804→                  const isExpanded = expandedId === n.id;
   805→                  const message = n.body || "";
   806→                  const isLong = message.length > 80;
   807→                  return (
   808→                    <div
   809→                      key={n.id}
   810→                      className={cn(
   811→                        "rounded-lg border p-2.5 transition-colors",
   812→                        !n.read ? "border-primary/20 bg-primary/5" : "border-transparent hover:bg-accent",
   813→                      )}
   814→                    >
   815→                      <div className="flex items-start gap-2">
   816→                        {!n.read && <span className="mt-1.5 h-2 w-2 shrink-0 rounded-full bg-primary" />}
   817→                        <div className="min-w-0 flex-1">
   818→                          <p className="text-xs font-semibold leading-tight">{n.title}</p>
   819→                          <p className={cn("mt-0.5 text-[11px] leading-snug text-muted-foreground", !isExpanded && isLong && "line-clamp-2")}>
   820→                            {message}
   821→                          </p>
   822→                          <div className="mt-1 flex items-center gap-2">
   823→                            <span className="text-[10px] text-muted-foreground/70">
   824→                              {new Date(n.createdAt).toLocaleDateString("en-NG", { month: "short", day: "numeric" })} ·{" "}
   825→                              {new Date(n.createdAt).toLocaleTimeString("en-NG", { hour: "2-digit", minute: "2-digit" })}
   826→                            </span>
   827→                            {isLong && (
   828→                              <button
   829→                                onClick={() => setExpandedId(isExpanded ? null : n.id)}
   830→                                className="text-[10px] font-medium text-primary hover:underline"
   831→                              >
   832→                                {isExpanded ? "See less" : "See more"}
   833→                              </button>
   834→                            )}
   835→                            {!n.read && (
   836→                              <button
   837→                                onClick={() => markOneRead(n.id)}
   838→                                className="text-[10px] font-medium text-muted-foreground hover:text-primary"
   839→                              >
   840→                                Mark read
   841→                              </button>
   842→                            )}
   843→                          </div>
   844→                        </div>
   845→                      </div>
   846→                    </div>
   847→                  );
   848→                })}
   849→              </div>
   850→            ) : (
   851→              <div className="py-8 text-center text-sm text-muted-foreground">No notifications yet.</div>
   852→            )}
   853→          </div>
   854→        </div>
   855→      )}
   856→    </>
   857→  );
   858→}
