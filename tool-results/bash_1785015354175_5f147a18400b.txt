   121→  onClick,

   124→  onClick?: () => void;

   133→    <button
   134→      onClick={onClick}
   135→      className="group block w-full text-left"

   174→        {/* status pill */}

   178→              card.status === "ACTIVE"
   179→                ? "bg-emerald-400/30 text-emerald-50"
   180→                : card.status === "FROZEN"
   181→                ? "bg-amber-400/30 text-amber-50"

   211→export default function CardsView() {
   212→  const { user, setView } = useApp();
   213→  const pin = usePin();
   214→  const [data, setData] = React.useState<CardsData | null>(null);
   215→  const [loading, setLoading] = React.useState(true);
   216→  const [busy, setBusy] = React.useState(false);
   217→
   218→  // create dialog
   219→  const [createOpen, setCreateOpen] = React.useState(false);
   220→  const [cardholder, setCardholder] = React.useState(user?.fullName ?? "");
   221→  const [creating, setCreating] = React.useState(false);
   222→
   223→  // generic amount dialog (fund/withdraw)
   224→  const [amountOpen, setAmountOpen] = React.useState(false);
   225→  const [amountMode, setAmountMode] = React.useState<"FUND" | "WITHDRAW">("FUND");
   226→  const [activeCard, setActiveCard] = React.useState<VCard | null>(null);
   227→  const [amountInput, setAmountInput] = React.useState("");
   228→  const amountKobo = parseKobo(amountInput);
   229→
   230→  // reveal dialog
   231→  const [revealOpen, setRevealOpen] = React.useState(false);
   232→  const [revealData, setRevealData] = React.useState<{ pan: string; cvv: string; expiry: string; cardholder: string; brand: string } | null>(null);
   233→  const [revealCountdown, setRevealCountdown] = React.useState(30);
   234→  const [revealCopied, setRevealCopied] = React.useState<string | null>(null);
   235→
   236→  // transactions dialog
   237→  const [txOpen, setTxOpen] = React.useState(false);
   238→  const [txCard, setTxCard] = React.useState<VCard | null>(null);
   239→
   240→  // terminate confirm
   241→  const [terminateOpen, setTerminateOpen] = React.useState(false);
   242→  const [terminateCard, setTerminateCard] = React.useState<VCard | null>(null);
   243→
   244→  const load = React.useCallback(async () => {
   245→    setLoading(true);
   246→    try {
   247→      const res = await fetch("/api/cards", { cache: "no-store" });
   248→      if (res.ok) setData(await res.json());
   249→      else if (res.status === 401) toast.error("Session expired. Please log in again.");
   250→      else toast.error("Failed to load cards.");
   251→    } finally {
   252→      setLoading(false);
   253→    }
   254→  }, []);
   255→
   256→  React.useEffect(() => {
   257→    load();
   258→  }, [load]);
   259→
   260→  React.useEffect(() => {
   261→    if (user?.fullName) setCardholder(user.fullName);
   262→  }, [user?.fullName]);
   263→
   264→  // reveal countdown
   265→  React.useEffect(() => {
   266→    if (!revealOpen) return;
   267→    setRevealCountdown(30);
   268→    const t = setInterval(() => {
   269→      setRevealCountdown((c) => {
   270→        if (c <= 1) {
   271→          clearInterval(t);
   272→          setRevealOpen(false);
   273→          setRevealData(null);
   274→          return 0;
   275→        }
   276→        return c - 1;
   277→      });
   278→    }, 1000);
   279→    return () => clearInterval(t);
   280→  }, [revealOpen]);
   281→
   282→  async function createCard() {
   283→    if (cardholder.trim().length < 2) {
   284→      toast.error("Enter a valid cardholder name");
   285→      return;
   286→    }
   287→    setCreating(true);
   288→    try {
   289→      const res = await fetch("/api/cards", {
   290→        method: "POST",
   291→        headers: { "Content-Type": "application/json" },
   292→        body: JSON.stringify({ cardholder: cardholder.trim() }),
   293→      });
   294→      const j = await res.json();
   295→      if (!res.ok) {
   296→        toast.error(j?.error ?? "Failed to create card");
   297→        return;
   298→      }
   299→      toast.success("Virtual card created");
   300→      setCreateOpen(false);
   301→      setCardholder(user?.fullName ?? "");
   302→      load();
   303→    } finally {
   304→      setCreating(false);
   305→    }
   306→  }
   307→
   308→  function openAmount(card: VCard, mode: "FUND" | "WITHDRAW") {
   309→    if (card.status === "TERMINATED") {
   310→      toast.error("Card is terminated");
   311→      return;
   312→    }
   313→    if (card.status === "FROZEN") {
   314→      toast.error(`Unfreeze the card to ${mode.toLowerCase()}`);
   315→      return;
   316→    }
   317→    setActiveCard(card);
   318→    setAmountMode(mode);
   319→    setAmountInput("");
   320→    setAmountOpen(true);
   321→  }
   322→
   323→  async function submitAmount() {
   324→    if (!activeCard) return;
   325→    if (amountKobo <= 0) {
   326→      toast.error("Enter a valid amount");
   327→      return;
   328→    }
   329→    const pinVal = await pin.request({
   330→      title: amountMode === "FUND" ? "Fund card" : "Withdraw from card",
   331→      description: `${naira(amountKobo)} · Card •••• ${activeCard.last4}`,
   332→    });
   333→    if (!pinVal) return;
   334→
   335→    setBusy(true);
   336→    try {
   337→      const endpoint =
   338→        amountMode === "FUND"
   339→          ? `/api/cards/${activeCard.id}/fund`
   340→          : `/api/cards/${activeCard.id}/withdraw`;
   341→      const res = await fetch(endpoint, {
   342→        method: "POST",
   343→        headers: { "Content-Type": "application/json" },
   344→        body: JSON.stringify({ amountKobo, pin: pinVal }),
   345→      });
   346→      const j = await res.json();
   347→      if (!res.ok) {
   348→        toast.error(j?.error ?? "Transaction failed");
   349→        return;
   350→      }
   351→      toast.success(
   352→        amountMode === "FUND"
   353→          ? `Card funded with ${naira(amountKobo)}`
   354→          : `${naira(amountKobo)} withdrawn to wallet`,
   355→      );
   356→      setAmountOpen(false);
   357→      setAmountInput("");
   358→      load();
   359→    } finally {
   360→      setBusy(false);
   361→    }
   362→  }
   363→
   364→  async function toggleFreeze(card: VCard) {
   365→    if (card.status === "TERMINATED") return;
   366→    const action = card.status === "FROZEN" ? "unfreeze" : "freeze";
   367→    setBusy(true);
   368→    try {
   369→      const res = await fetch(`/api/cards/${card.id}`, {
   370→        method: "PATCH",
   371→        headers: { "Content-Type": "application/json" },
   372→        body: JSON.stringify({ action }),
   373→      });
   374→      const j = await res.json();
   375→      if (!res.ok) {
   376→        toast.error(j?.error ?? "Action failed");
   377→        return;
   378→      }
   379→      toast.success(action === "freeze" ? "Card frozen" : "Card reactivated");
   380→      load();
   381→    } finally {
   382→      setBusy(false);
   383→    }
   384→  }
   385→
   386→  function promptTerminate(card: VCard) {
   387→    setTerminateCard(card);
   388→    setTerminateOpen(true);
   389→  }
   390→
   391→  async function doTerminate() {
   392→    if (!terminateCard) return;
   393→    setBusy(true);
   394→    try {
   395→      const res = await fetch(`/api/cards/${terminateCard.id}`, {
   396→        method: "PATCH",
   397→        headers: { "Content-Type": "application/json" },
   398→        body: JSON.stringify({ action: "terminate" }),
   399→      });
   400→      const j = await res.json();
   401→      if (!res.ok) {
   402→        toast.error(j?.error ?? "Termination failed");
   403→        return;
   404→      }
   405→      toast.success("Card terminated");
   406→      setTerminateOpen(false);
   407→      setTerminateCard(null);
   408→      load();
   409→    } finally {
   410→      setBusy(false);
   411→    }
   412→  }
   413→
   414→  async function reveal(card: VCard) {
   415→    if (card.status === "TERMINATED") {
   416→      toast.error("Card is terminated");
   417→      return;
   418→    }
   419→    setBusy(true);
   420→    try {
   421→      const res = await fetch(`/api/cards/${card.id}/reveal`, {
   422→        method: "POST",
   423→        headers: { "Content-Type": "application/json" },
   424→      });
   425→      const j = await res.json();
   426→      if (!res.ok) {
   427→        toast.error(j?.error ?? "Reveal failed");
   428→        return;
   429→      }
   430→      setRevealData(j);
   431→      setRevealOpen(true);
   432→    } finally {
   433→      setBusy(false);
   434→    }
   435→  }
   436→
   437→  function showTransactions(card: VCard) {
   438→    setTxCard(card);
   439→    setTxOpen(true);
   440→  }
   441→
   442→  function copyReveal(field: string, value: string) {
   443→    navigator.clipboard.writeText(value);
   444→    setRevealCopied(field);
   445→    toast.success(`${field} copied`);
   446→    setTimeout(() => setRevealCopied(null), 1500);
   447→  }
   448→
   449→  const cards = data?.cards ?? [];
   450→
   451→  return (
   452→    <div className="space-y-6 tp-fade-rise">
   453→      <PageHeader
   454→        title="Virtual Cards"
   455→        subtitle="Spend anywhere Visa or Mastercard is accepted."
   456→        actions={
   457→          <>
   458→            <Button variant="outline" size="sm" onClick={load} disabled={loading} className="gap-1.5">
   459→              <RefreshCw className={loading ? "h-4 w-4 animate-spin" : "h-4 w-4"} /> Refresh
   460→            </Button>
   461→            <FeatureGate
   462→              requiredTier={2}
   463→              feature="Virtual cards"
   464→              compact
   465→              fallback={
   466→                <Button
   467→                  size="sm"
   468→                  variant="outline"
   469→                  onClick={() => setView("kyc")}
   470→                  className="gap-1.5 border-amber-500/40 bg-amber-500/10 text-amber-700 hover:bg-amber-500/15 dark:text-amber-400"
   471→                >
   472→                  <Lock className="h-4 w-4" /> Upgrade to create
   473→                </Button>
   474→              }
   475→            >
   476→              <Button size="sm" onClick={() => setCreateOpen(true)} className="gap-1.5">
   477→                <Plus className="h-4 w-4" /> Create card
   478→              </Button>
   479→            </FeatureGate>
   480→          </>
   481→        }
   482→      />
   483→
   484→      {loading ? (
   485→        <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-3">
   486→          {[0, 1, 2].map((i) => (
   487→            <Skeleton key={i} className="aspect-[1.586/1] w-full rounded-2xl" />
   488→          ))}
   489→        </div>
   490→      ) : cards.length === 0 ? (
   491→        <FeatureGate
   492→          requiredTier={2}
   493→          feature="Virtual cards"
   494→          description="Verify your NIN (KYC Tier 2) to issue virtual cards you can spend anywhere Visa or Mastercard is accepted."
   495→        >
   496→          <EmptyState
   497→            icon={CreditCard}
   498→            title="No virtual cards yet"
   499→            description="Create a card to spend online or in-store. Cards are issued instantly."
   500→            action={
   501→              <Button size="sm" onClick={() => setCreateOpen(true)} className="gap-1.5">
   502→                <Plus className="h-4 w-4" /> Create your first card
   503→              </Button>
   504→            }
   505→          />
   506→        </FeatureGate>
   507→      ) : (
   508→        <div className="grid gap-6 sm:grid-cols-2 lg:grid-cols-3">
   509→          {cards.map((c) => (
   510→            <div key={c.id} className="space-y-2">
   511→              <CardVisual card={c} onClick={() => {}} />
   512→              <div className="flex items-center justify-between gap-2 px-1">
   513→                <Badge variant="outline" className={`gap-1 ${STATUS_TONE[c.status] ?? ""}`}>
   514→                  {c.status}
   515→                </Badge>
   516→                <DropdownMenu>
   517→                  <DropdownMenuTrigger asChild>
   518→                    <Button variant="ghost" size="sm" className="gap-1" disabled={busy}>
   519→                      <MoreHorizontal className="h-4 w-4" /> Actions
   520→                    </Button>
   521→                  </DropdownMenuTrigger>
   522→                  <DropdownMenuContent align="end" className="w-48">
   523→                    <DropdownMenuItem onClick={() => openAmount(c, "FUND")}>
   524→                      <ArrowDownToLine className="mr-2 h-4 w-4 text-emerald-600" /> Fund card
   525→                    </DropdownMenuItem>
   526→                    <DropdownMenuItem onClick={() => openAmount(c, "WITHDRAW")}>
   527→                      <ArrowUpFromLine className="mr-2 h-4 w-4 text-amber-600" /> Withdraw
   528→                    </DropdownMenuItem>
   529→                    <DropdownMenuItem onClick={() => reveal(c)}>
   530→                      <Eye className="mr-2 h-4 w-4" /> Reveal details
   531→                    </DropdownMenuItem>
   532→                    <DropdownMenuItem onClick={() => showTransactions(c)}>
   533→                      <History className="mr-2 h-4 w-4" /> Transactions
   534→                    </DropdownMenuItem>
   535→                    <DropdownMenuSeparator />
   536→                    <DropdownMenuItem onClick={() => toggleFreeze(c)}>
   537→                      {c.status === "FROZEN" ? (
   538→                        <>
   539→                          <Sun className="mr-2 h-4 w-4 text-amber-600" /> Unfreeze
   540→                        </>
   541→                      ) : (
   542→                        <>
   543→                          <Snowflake className="mr-2 h-4 w-4 text-sky-600" /> Freeze
   544→                        </>
   545→                      )}
   546→                    </DropdownMenuItem>
   547→                    <DropdownMenuItem
   548→                      className="text-red-600 focus:text-red-700"
   549→                      onClick={() => promptTerminate(c)}
   550→                      disabled={c.status === "TERMINATED"}
   551→                    >
   552→                      <Ban className="mr-2 h-4 w-4" /> Terminate
   553→                    </DropdownMenuItem>
   554→                  </DropdownMenuContent>
   555→                </DropdownMenu>
   556→              </div>
   557→            </div>
   558→          ))}
   559→        </div>
   560→      )}
   561→
   562→      {/* Create card dialog */}
   563→      <Dialog open={createOpen} onOpenChange={setCreateOpen}>
   564→        <DialogContent className="max-w-md">
   565→          <DialogHeader>
   566→            <DialogTitle>Create a virtual card</DialogTitle>
   567→            <DialogDescription>
   568→              Instant issuance. Card number is encrypted at rest. Default limit is ₦5,000 — fund the card from your wallet to spend.
   569→            </DialogDescription>
   570→          </DialogHeader>
   571→          <div className="space-y-3 py-1">
   572→            <div className="space-y-2">
   573→              <Label htmlFor="cardholder">Cardholder name</Label>
   574→              <Input
   575→                id="cardholder"
   576→                value={cardholder}
   577→                onChange={(e) => setCardholder(e.target.value)}
   578→                placeholder="JOHN DOE"
   579→                className="uppercase"
   580→                maxLength={40}
   581→              />
   582→            </div>
   583→            <div className="rounded-xl border bg-muted/40 p-3 text-xs text-muted-foreground">
   584→              <p className="font-medium text-foreground">Heads up</p>
   585→              <ul className="mt-1 list-disc space-y-1 pl-4">
   586→                <li>A 16-digit PAN, CVV and expiry will be generated.</li>
   587→                <li>Brand (Visa / Mastercard) is selected automatically.</li>
   588→                <li>You can reveal the full PAN any time — every reveal is audited.</li>
   589→              </ul>
   590→            </div>
   591→          </div>
   592→          <DialogFooter>
   593→            <Button variant="ghost" onClick={() => setCreateOpen(false)}>Cancel</Button>
   594→            <Button onClick={createCard} disabled={creating} className="gap-1.5">
   595→              {creating ? <RefreshCw className="h-4 w-4 animate-spin" /> : <Plus className="h-4 w-4" />}
   596→              Create card
   597→            </Button>
   598→          </DialogFooter>
   599→        </DialogContent>
   600→      </Dialog>
   601→
   602→      {/* Fund / Withdraw dialog */}
   603→      <Dialog open={amountOpen} onOpenChange={(o) => !busy && setAmountOpen(o)}>
   604→        <DialogContent className="max-w-md">
   605→          <DialogHeader>
   606→            <DialogTitle>
   607→              {amountMode === "FUND" ? "Fund card" : "Withdraw from card"}
   608→            </DialogTitle>
   609→            <DialogDescription>
   610→              {amountMode === "FUND"
   611→                ? `Move money from your wallet to card •••• ${activeCard?.last4 ?? ""}.`
   612→                : `Move money from card •••• ${activeCard?.last4 ?? ""} back to your wallet.`}
   613→            </DialogDescription>
   614→          </DialogHeader>
   615→          <div className="space-y-3 py-1">
   616→            <div className="space-y-2">
   617→              <Label htmlFor="amt">Amount (₦)</Label>
   618→              <Input
   619→                id="amt"
   620→                inputMode="numeric"
   621→                placeholder="0.00"
   622→                value={amountInput}
   623→                onChange={(e) => setAmountInput(e.target.value)}
   624→              />
   625→              <div className="flex flex-wrap gap-1.5">
   626→                {QUICK_AMOUNTS.map((a) => (
   627→                  <button
   628→                    key={a.label}
   629→                    type="button"
   630→                    onClick={() => setAmountInput(String(a.kobo / 100))}
   631→                    className="rounded-full border border-border bg-background px-2.5 py-1 text-xs font-medium hover:border-primary hover:bg-primary/5"
   632→                  >
   633→                    {a.label}
   634→                  </button>
   635→                ))}
   636→              </div>
   637→            </div>
   638→            {amountKobo > 0 && (
   639→              <div className="rounded-xl border bg-muted/40 p-3 text-sm">
   640→                <div className="flex items-center justify-between">
   641→                  <span className="text-muted-foreground">
   642→                    {amountMode === "FUND" ? "Card balance after" : "Card balance after"}
   643→                  </span>
   644→                  <span className="font-semibold tabular-nums">
   645→                    {naira(
   646→                      (activeCard?.balanceKobo ?? 0) +
   647→                        (amountMode === "FUND" ? amountKobo : -amountKobo),
   648→                    )}
   649→                  </span>
   650→                </div>
   651→              </div>
   652→            )}
   653→          </div>
   654→          <DialogFooter>
   655→            <Button variant="ghost" onClick={() => setAmountOpen(false)} disabled={busy}>Cancel</Button>
   656→            <Button onClick={submitAmount} disabled={busy || amountKobo <= 0} className="gap-1.5">
   657→              {busy ? <RefreshCw className="h-4 w-4 animate-spin" /> : <Check className="h-4 w-4" />}
   658→              Continue
   659→            </Button>
   660→          </DialogFooter>
   661→        </DialogContent>
   662→      </Dialog>
   663→
   664→      {/* Reveal dialog */}
   665→      <Dialog open={revealOpen} onOpenChange={(o) => !busy && setRevealOpen(o)}>
   666→        <DialogContent className="max-w-md">
   667→          <DialogHeader>
   668→            <DialogTitle>Card details</DialogTitle>
   669→            <DialogDescription>
   670→              For your security, these details auto-hide in {revealCountdown}s. Never share them.
   671→            </DialogDescription>
   672→          </DialogHeader>
   673→          {revealData && (
   674→            <div className="space-y-3 py-1">
   675→              <div className="rounded-2xl bg-gradient-to-br from-slate-800 via-slate-700 to-slate-900 p-4 text-white">
   676→                <div className="flex items-start justify-between">
   677→                  <p className="text-[10px] uppercase tracking-widest opacity-70">{revealData.brand}</p>
   678→                  <BrandLogo brand={revealData.brand} />
   679→                </div>
   680→                <p className="mt-4 font-mono text-lg tracking-wider">{revealData.pan}</p>
   681→                <div className="mt-3 flex justify-between text-xs">
   682→                  <div>
   683→                    <p className="text-[9px] uppercase opacity-70">Cardholder</p>
   684→                    <p className="font-medium uppercase">{revealData.cardholder}</p>
   685→                  </div>
   686→                  <div className="text-right">
   687→                    <p className="text-[9px] uppercase opacity-70">EXP / CVV</p>
   688→                    <p className="font-mono">{revealData.expiry} · {revealData.cvv}</p>
   689→                  </div>
   690→                </div>
   691→              </div>
   692→              <div className="grid grid-cols-2 gap-2">
   693→                <Button variant="outline" size="sm" className="gap-1.5" onClick={() => copyReveal("PAN", revealData.pan)}>
   694→                  {revealCopied === "PAN" ? <Check className="h-3.5 w-3.5 text-emerald-600" /> : <Copy className="h-3.5 w-3.5" />}
   695→                  Copy PAN
   696→                </Button>
   697→                <Button variant="outline" size="sm" className="gap-1.5" onClick={() => copyReveal("CVV", revealData.cvv)}>
   698→                  {revealCopied === "CVV" ? <Check className="h-3.5 w-3.5 text-emerald-600" /> : <Copy className="h-3.5 w-3.5" />}
   699→                  Copy CVV
   700→                </Button>
   701→              </div>
   702→              <div className="flex items-center gap-2 rounded-xl border border-amber-500/30 bg-amber-500/10 p-3 text-xs text-amber-700 dark:text-amber-300">
   703→                <ShieldAlert className="h-4 w-4 shrink-0" />
   704→                <span>This reveal has been recorded in your audit log.</span>
   705→              </div>
   706→              <Progress value={(revealCountdown / 30) * 100} className="h-1" />
   707→            </div>
   708→          )}
   709→          <DialogFooter>
   710→            <Button variant="ghost" onClick={() => setRevealOpen(false)} className="gap-1.5">
   711→              <EyeOff className="h-4 w-4" /> Hide now
   712→            </Button>
   713→          </DialogFooter>
   714→        </DialogContent>
   715→      </Dialog>
   716→
   717→      {/* Transactions dialog */}
   718→      <Dialog open={txOpen} onOpenChange={setTxOpen}>
   719→        <DialogContent className="max-w-md">
   720→          <DialogHeader>
   721→            <DialogTitle>Card transactions</DialogTitle>
   722→            <DialogDescription>
   723→              Card •••• {txCard?.last4 ?? ""} · last {txCard?.recentTransactions.length ?? 0} transactions.
   724→            </DialogDescription>
   725→          </DialogHeader>
   726→          <div className="max-h-96 overflow-y-auto scrollbar-thin">
   727→            {txCard && txCard.recentTransactions.length > 0 ? (
   728→              <div className="space-y-1">
   729→                {txCard.recentTransactions.map((t) => {
   730→                  const isCredit = t.type === "FUND";
   731→                  return (
   732→                    <div
   733→                      key={t.id}
   734→                      className="flex items-center gap-3 rounded-xl px-2 py-2.5 hover:bg-muted/60"
   735→                    >
   736→                      <div
   737→                        className={`flex h-9 w-9 shrink-0 items-center justify-center rounded-full ${
   738→                          isCredit
   739→                            ? "bg-emerald-500/15 text-emerald-600 dark:text-emerald-400"
   740→                            : "bg-amber-500/15 text-amber-600 dark:text-amber-400"
   741→                        }`}
   742→                      >
   743→                        {isCredit ? (
   744→                          <ArrowDownToLine className="h-4 w-4" />
   745→                        ) : (
   746→                          <ArrowUpFromLine className="h-4 w-4" />
   747→                        )}
   748→                      </div>
   749→                      <div className="min-w-0 flex-1">
   750→                        <p className="truncate text-sm font-medium">{t.description || t.type}</p>
   751→                        <p className="truncate text-xs text-muted-foreground">
   752→                          {t.type} · {timeAgo(t.createdAt)}
   753→                        </p>
   754→                      </div>
   755→                      <p
   756→                        className={`text-sm font-semibold tabular-nums ${
   757→                          isCredit ? "text-emerald-600 dark:text-emerald-400" : "text-foreground"
   758→                        }`}
   759→                      >
   760→                        {isCredit ? "+" : "−"}
   761→                        {naira(t.amountKobo)}
   762→                      </p>
   763→                    </div>
   764→                  );
   765→                })}
   766→              </div>
   767→            ) : (
   768→              <div className="py-10 text-center text-sm text-muted-foreground">
   769→                No transactions yet.
   770→              </div>
   771→            )}
   772→          </div>
   773→        </DialogContent>
   774→      </Dialog>
   775→
   776→      {/* Terminate confirm */}
   777→      <AlertDialog open={terminateOpen} onOpenChange={(o) => !busy && setTerminateOpen(o)}>
   778→        <AlertDialogContent>
   779→          <AlertDialogHeader>
   780→            <AlertDialogTitle>Terminate card •••• {terminateCard?.last4 ?? ""}?</AlertDialogTitle>
   781→            <AlertDialogDescription>
   782→              This action cannot be undone. The card will be permanently disabled. Any remaining balance should be withdrawn first.
   783→            </AlertDialogDescription>
   784→          </AlertDialogHeader>
   785→          <AlertDialogFooter>
   786→            <AlertDialogCancel disabled={busy}>Cancel</AlertDialogCancel>
   787→            <AlertDialogAction
   788→              onClick={doTerminate}
   789→              disabled={busy}
   790→              className="bg-red-600 text-white hover:bg-red-700"
   791→            >
   792→              {busy ? <RefreshCw className="h-4 w-4 animate-spin" /> : <Ban className="h-4 w-4" />}
   793→              Terminate card
   794→            </AlertDialogAction>
   795→          </AlertDialogFooter>
   796→        </AlertDialogContent>
   797→      </AlertDialog>
   798→    </div>
   799→  );
   800→}
