   349→  async function generateStatement() {
   350→    if (!stmtStart || !stmtEnd) {
   351→      toast.error("Please pick a start and end date");
   352→      return;
   353→    }
   354→    const start = new Date(stmtStart);
   355→    start.setHours(0, 0, 0, 0);
   356→    const end = new Date(stmtEnd);
   357→    end.setHours(23, 59, 59, 999);
   358→    if (start >= end) {
   359→      toast.error("Start date must be before end date");
   360→      return;
   361→    }
   362→    setGenerating(true);
   363→    try {
   364→      const res = await fetch("/api/statements", {
   365→        method: "POST",
   366→        headers: { "Content-Type": "application/json" },
   367→        body: JSON.stringify({
   368→          periodStart: start.toISOString(),
   369→          periodEnd: end.toISOString(),
   370→          format: stmtFormat,
   371→        }),
   372→      });
   373→      const data = await res.json();
   374→      if (!res.ok) {
   375→        toast.error(data.error ?? "Failed to generate statement");
   376→        return;
   377→      }
   378→      toast.success(
   379→        `Statement ready — ${data.statement.transactionCount ?? 0} transactions`,
   380→      );
   381→      setStmtOpen(false);
   382→      // Auto-download
   383→      const a = document.createElement("a");
   384→      a.href = data.statement.downloadUrl;
   385→      a.download = "";
   386→      document.body.appendChild(a);
   387→      a.click();
   388→      a.remove();
   389→    } catch (e) {
   390→      toast.error(e instanceof Error ? e.message : "Failed to generate statement");
   391→    } finally {
   392→      setGenerating(false);
   393→    }
   394→  }

   396→  async function handleExport() {
   397→    setExporting(true);
   398→    try {
   399→      // Pull last 90 days (server returns up to 100, paginated)
   400→      const params = new URLSearchParams({
   401→        page: "1",
   402→        limit: "100",
   403→      });
   404→      if (filter) params.set("filter", filter);
   405→      if (debouncedSearch) params.set("search", debouncedSearch);
   406→      if (appliedTypes.length > 0) params.set("type", appliedTypes.join(","));
   407→      if (appliedStatus !== "ALL") params.set("status", appliedStatus);
   408→      if (appliedDirection !== "ALL") params.set("direction", appliedDirection);
   409→      if (appliedMin) params.set("minAmount", appliedMin);
   410→      if (appliedMax) params.set("maxAmount", appliedMax);
   411→      if (appliedDateFrom) params.set("dateFrom", appliedDateFrom);
   412→      if (appliedDateTo) params.set("dateTo", appliedDateTo);
   413→      const res = await fetch(`/api/transactions?${params.toString()}`, { cache: "no-store" });
   414→      if (!res.ok) throw new Error("Failed to fetch transactions");
   415→      const data: TxResponse = await res.json();
   416→
   417→      const since = new Date(Date.now() - 90 * 24 * 60 * 60 * 1000);
   418→      const rows = data.transactions.filter((t) => new Date(t.createdAt) >= since);
   419→      if (rows.length === 0) {
   420→        toast.error("No transactions in the last 90 days to export");
   421→        return;
   422→      }
   423→
   424→      const headers = [
   425→        "Date",
   426→        "Reference",
   427→        "Type",
   428→        "Direction",
   429→        "Counterparty",
   430→        "Description",
   431→        "Amount (NGN)",
   432→        "Fee (NGN)",
   433→        "Status",
   434→        "Provider",
   435→      ];
   436→      const escape = (v: string) => `"${(v ?? "").replace(/"/g, '""')}"`;
   437→      const lines = [headers.join(",")];
   438→      for (const t of rows) {
   439→        lines.push(
   440→          [
   441→            new Date(t.createdAt).toISOString(),
   442→            t.reference,
   443→            t.type,
   444→            t.direction,
   445→            t.counterpartyName ?? "",
   446→            t.description ?? "",
   447→            (t.amountKobo / 100).toFixed(2),
   448→            ((t.feeKobo ?? 0) / 100).toFixed(2),
   449→            t.status,
   450→            t.provider ?? "",
   451→          ]
   452→            .map((v) => escape(String(v)))
   453→            .join(","),
   454→        );
   455→      }
   456→      const csv = lines.join("\n");
   457→      const blob = new Blob([csv], { type: "text/csv;charset=utf-8" });
   458→      const url = URL.createObjectURL(blob);
   459→      const a = document.createElement("a");
   460→      a.href = url;
   461→      a.download = `turbopay-transactions-${new Date().toISOString().slice(0, 10)}.csv`;
   462→      document.body.appendChild(a);
   463→      a.click();
   464→      a.remove();
   465→      URL.revokeObjectURL(url);
   466→      toast.success(`Exported ${rows.length} transactions`);
   467→    } catch (e) {
   468→      toast.error(e instanceof Error ? e.message : "Export failed");
   469→    } finally {
   470→      setExporting(false);
   471→    }
   472→  }

   571→      {/* Advanced filter panel */}
   572→      <Collapsible open={advancedOpen} onOpenChange={setAdvancedOpen}>

   914→function TxDetailDialog({ tx, onClose }: { tx: Tx | null; onClose: () => void }) {
   915→  const open = !!tx;
   916→  if (!tx) return <Dialog open={open} onOpenChange={onClose} />;
   917→
   918→  const isCredit = tx.direction === "CREDIT";
   919→  const tone = STATUS_TONE[tx.status] ?? STATUS_TONE.PENDING;
   920→
   921→  return (
   922→    <Dialog open={open} onOpenChange={onClose}>
   923→      <DialogContent className="max-w-md">
   924→        <DialogHeader>
   925→          <div
   926→            className={`mx-auto flex h-14 w-14 items-center justify-center rounded-full ${
   927→              isCredit
   928→                ? "bg-emerald-500/15 text-emerald-600 dark:text-emerald-400"
   929→                : "bg-amber-500/15 text-amber-600 dark:text-amber-400"
   930→            }`}
   931→          >
   932→            <span className="text-2xl font-bold">{isCredit ? "+" : "−"}</span>
   933→          </div>
   934→          <DialogTitle className="text-center">
   935→            {naira(tx.amountKobo)}
   936→          </DialogTitle>
   937→          <DialogDescription className="text-center">
   938→            {TYPE_LABELS[tx.type] ?? tx.type}
   939→          </DialogDescription>
   940→        </DialogHeader>
   941→
   942→        <div className="flex justify-center">
   943→          <span className={`rounded-full px-2.5 py-1 text-xs font-semibold ${tone}`}>
   944→            {tx.status}
   945→          </span>
   946→        </div>
   947→
   948→        <div className="space-y-2.5 rounded-xl bg-muted/40 p-4 text-sm">
   949→          <DetailRow label="Reference" value={tx.reference} mono />
   950→          {tx.description && <DetailRow label="Description" value={tx.description} />}

   952→            <DetailRow label="Counterparty" value={tx.counterpartyName} />
