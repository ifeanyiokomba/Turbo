    51→
    52→type TransferType = "TURBOPAY" | "BANK";
    53→
    54→interface Beneficiary {
    55→  id: string;
    56→  name: string;
    57→  accountNumber: string;
    58→  bankName: string;
    59→  bankCode: string | null;
    60→  type: string;
    61→  isFavorite: boolean;
    62→  lastUsedAt: string | null;
    63→}
    64→
    65→interface ResolveResult {
    66→  name: string;
    67→  type: "TURBOPAY" | "BANK";
    68→  accountNumber?: string;
    69→  bankName?: string;
    70→  username?: string;
    71→}
    72→
    73→interface TransferResult {
    74→  transaction: {
    75→    id: string;
    76→    reference: string;
    77→    amountKobo: number;
    78→    feeKobo: number;
    79→    counterpartyName: string | null;
    80→    counterpartyAccount: string | null;
    81→    counterpartyBank: string | null;
    82→    createdAt: string;
    83→  };
    84→  newBalance: number;
    85→  type: TransferType;
    86→  recipientName: string;
    87→}
    88→
    89→const BANK_FEE_KOBO = 5250;
    90→
    91→export default function TransferView() {
    92→  const { setView } = useApp();
    93→  const pin = usePin();
    94→
    95→  const [type, setType] = React.useState<TransferType>("TURBOPAY");
    96→
    97→  // Turbopay form
    98→  const [tpRecipient, setTpRecipient] = React.useState("");
    99→  const [tpResolved, setTpResolved] = React.useState<ResolveResult | null>(null);
   100→  const [tpResolving, setTpResolving] = React.useState(false);
   101→
   102→  // Bank form
   103→  const [bankCode, setBankCode] = React.useState("");
   104→  const [bankAccount, setBankAccount] = React.useState("");
   105→  const [bankResolved, setBankResolved] = React.useState<ResolveResult | null>(null);
   106→  const [bankResolving, setBankResolving] = React.useState(false);
   107→
   108→  // Shared fields
   109→  const [amountInput, setAmountInput] = React.useState("");
   110→  const [note, setNote] = React.useState("");
   111→  const [saveBeneficiary, setSaveBeneficiary] = React.useState(true);
   112→
   113→  // Beneficiaries list
   114→  const [beneficiaries, setBeneficiaries] = React.useState<Beneficiary[]>([]);
   115→  const [benLoading, setBenLoading] = React.useState(true);
   116→
   117→  // Confirm + success
   118→  const [confirmOpen, setConfirmOpen] = React.useState(false);
   119→  const [submitting, setSubmitting] = React.useState(false);
   120→  const [result, setResult] = React.useState<TransferResult | null>(null);
   121→
   122→  const amountKobo = parseKobo(amountInput);
   123→  const feeKobo = type === "BANK" ? BANK_FEE_KOBO : 0;
   124→  const totalKobo = amountKobo + feeKobo;
   125→
   126→  const resolved = type === "TURBOPAY" ? tpResolved : bankResolved;
   127→  const recipientLabel = type === "TURBOPAY" ? tpRecipient : bankAccount;
   128→
   129→  const loadBeneficiaries = React.useCallback(async () => {
   130→    setBenLoading(true);
   131→    try {
   132→      const res = await fetch("/api/beneficiaries", { cache: "no-store" });
   133→      if (res.ok) {
   134→        const json = await res.json();
   135→        setBeneficiaries(json.beneficiaries ?? []);
   136→      }
   137→    } finally {
   138→      setBenLoading(false);
   139→    }
   140→  }, []);
   141→
   142→  React.useEffect(() => {
   143→    loadBeneficiaries();
   144→  }, [loadBeneficiaries]);
   145→
   146→  // Prefill from beneficiaries view "Send" action
   147→  React.useEffect(() => {
   148→    try {
   149→      const id = sessionStorage.getItem("tp_prefill_beneficiary");
   150→      if (!id) return;
   151→      sessionStorage.removeItem("tp_prefill_beneficiary");
   152→      (async () => {
   153→        try {
   154→          const res = await fetch("/api/beneficiaries", { cache: "no-store" });
   155→          if (!res.ok) return;
   156→          const json = await res.json();
   157→          const found: Beneficiary | undefined = (json.beneficiaries ?? []).find(
   158→            (b: Beneficiary) => b.id === id,
   159→          );
   160→          if (found) prefill(found);
   161→        } catch {}
   162→      })();
   163→    } catch {}
   164→  }, []);
   165→
   166→  function resetForm() {
   167→    setTpRecipient("");
   168→    setTpResolved(null);
   169→    setBankCode("");
   170→    setBankAccount("");
   171→    setBankResolved(null);
   172→    setAmountInput("");
   173→    setNote("");
   174→    setSaveBeneficiary(true);
   175→  }
   176→
   177→  async function resolveTurbopay() {
   178→    const q = tpRecipient.trim();
   179→    if (q.length < 3) {
   180→      toast.error("Enter a valid username, phone, email or account number");
   181→      return;
   182→    }
   183→    setTpResolving(true);
   184→    setTpResolved(null);
   185→    try {
   186→      const url = `/api/transfer/resolve?query=${encodeURIComponent(q)}`;
   187→      const res = await fetch(url);
   188→      const json = await res.json();
   189→      if (!res.ok) {
   190→        toast.error(json?.error ?? "Could not resolve recipient");
   191→        return;
   192→      }
   193→      setTpResolved(json);
   194→      toast.success(`Resolved: ${json.name}`);
   195→    } finally {
   196→      setTpResolving(false);
   197→    }
   198→  }
   199→
   200→  async function resolveBank() {
   201→    const acc = bankAccount.trim();
   202→    if (!/^\d{6,10}$/.test(acc)) {
   203→      toast.error("Enter a valid 10-digit account number");
   204→      return;
   205→    }
   206→    if (!bankCode) {
   207→      toast.error("Select a bank first");
   208→      return;
   209→    }
   210→    setBankResolving(true);
   211→    setBankResolved(null);
   212→    try {
   213→      const url = `/api/transfer/resolve?query=${encodeURIComponent(acc)}&bankCode=${encodeURIComponent(bankCode)}`;
   214→      const res = await fetch(url);
   215→      const json = await res.json();
   216→      if (!res.ok) {
   217→        toast.error(json?.error ?? "Could not resolve account");
   218→        return;
   219→      }
   220→      setBankResolved(json);
   221→      toast.success(`Resolved: ${json.name}`);
   222→    } finally {
   223→      setBankResolving(false);
   224→    }
   225→  }
   226→
   227→  function canContinue(): boolean {
   228→    if (amountKobo <= 0) return false;
   229→    if (type === "TURBOPAY") return !!tpResolved;
   230→    return !!bankResolved && !!bankCode;
   231→  }
   232→
   233→  function onContinue() {
   234→    if (!canContinue()) {
   235→      toast.error("Resolve recipient and enter amount first");
   236→      return;
   237→    }
   238→    setConfirmOpen(true);
   239→  }
   240→
   241→  async function submitTransfer() {
   242→    setConfirmOpen(false);
   243→    try {
   244→      const pinValue = await pin.request({
   245→        title: "Confirm transfer",
   246→        description: `Send ${naira(amountKobo)} to ${resolved?.name ?? recipientLabel}`,
   247→      });
   248→      if (!pinValue || pinValue.length !== 4) {
   249→        toast.error("PIN required to authorise transfer");
   250→        return;
   251→      }
   252→      setSubmitting(true);
   253→      const body = {
   254→        type,
   255→        recipient: type === "TURBOPAY" ? tpRecipient.trim() : bankAccount.trim(),
   256→        bankCode: type === "BANK" ? bankCode : undefined,
   257→        amountKobo,
   258→        note,
   259→        pin: pinValue,
   260→        saveBeneficiary,
   261→      };
   262→      const res = await fetch("/api/transfer", {
   263→        method: "POST",
   264→        headers: { "Content-Type": "application/json" },
   265→        body: JSON.stringify(body),
   266→      });
   267→      const json = await res.json();
   268→      if (!res.ok) {
   269→        toast.error(json?.error ?? "Transfer failed");
   270→        return;
   271→      }
   272→      setResult(json);
   273→      resetForm();
   274→      loadBeneficiaries();
   275→      toast.success("Transfer successful");
   276→    } catch {
   277→      toast.error("Network error. Try again.");
   278→    } finally {
   279→      setSubmitting(false);
   280→    }
   281→  }
   282→
   283→  async function toggleFavorite(b: Beneficiary) {
   284→    const next = !b.isFavorite;
   285→    setBeneficiaries((arr) =>
   286→      arr.map((x) => (x.id === b.id ? { ...x, isFavorite: next } : x)),
   287→    );
   288→    try {
   289→      await fetch(`/api/beneficiaries/${b.id}`, {
   290→        method: "PATCH",
   291→        headers: { "Content-Type": "application/json" },
   292→        body: JSON.stringify({ isFavorite: next }),
   293→      });
   294→    } catch {
   295→      // revert on failure
   296→      setBeneficiaries((arr) =>
   297→        arr.map((x) => (x.id === b.id ? { ...x, isFavorite: !next } : x)),
   298→      );
   299→    }
   300→  }
   301→
   302→  async function deleteBeneficiary(b: Beneficiary) {
   303→    const prev = beneficiaries;
   304→    setBeneficiaries((arr) => arr.filter((x) => x.id !== b.id));
   305→    try {
   306→      const res = await fetch(`/api/beneficiaries/${b.id}`, { method: "DELETE" });
   307→      if (!res.ok) throw new Error();
   308→      toast.success("Beneficiary removed");
   309→    } catch {
   310→      setBeneficiaries(prev);
   311→      toast.error("Could not delete beneficiary");
   312→    }
   313→  }
   314→
   315→  function prefill(b: Beneficiary) {
   316→    if (b.type === "TURBOPAY") {
   317→      setType("TURBOPAY");
   318→      setTpRecipient(b.accountNumber);
   319→      setTpResolved({
   320→        name: b.name,
   321→        type: "TURBOPAY",
   322→        accountNumber: b.accountNumber,
   323→        username: b.accountNumber,
   324→      });
   325→    } else {
   326→      setType("BANK");
   327→      setBankCode(b.bankCode ?? "");
   328→      setBankAccount(b.accountNumber);
   329→      setBankResolved({
   330→        name: b.name,
   331→        type: "BANK",
   332→        accountNumber: b.accountNumber,
   333→        bankName: b.bankName,
   334→      });
   335→    }
   336→    setAmountInput("");
   337→    setNote("");
   338→    setSaveBeneficiary(false);
   339→    toast.success(`Prefilled ${b.name}`);
   340→  }
   341→
   342→  return (
   343→    <div className="space-y-6 tp-fade-rise">
   344→      <PageHeader
   345→        title="Transfer"
   346→        subtitle="Send money to Turbopay users or any Nigerian bank account."
   347→      />
   348→
   349→      {result ? (
   350→        <SuccessCard result={result} onNew={() => setResult(null)} onViewHistory={() => setView("history")} />
   351→      ) : (
   352→        <div className="grid gap-6 lg:grid-cols-3">
   353→          {/* Left column - form */}
   354→          <div className="space-y-6 lg:col-span-2">
   355→            <Card className="p-5">
   356→              <Tabs value={type} onValueChange={(v) => setType(v as TransferType)}>
   357→                <TabsList className="grid w-full grid-cols-2">
   358→                  <TabsTrigger value="TURBOPAY" className="gap-1.5">
   359→                    <ArrowLeftRight className="h-3.5 w-3.5" /> Turbopay
   360→                  </TabsTrigger>
   361→                  <TabsTrigger value="BANK" className="gap-1.5">
   362→                    <Landmark className="h-3.5 w-3.5" /> Bank
   363→                  </TabsTrigger>
   364→                </TabsList>
   365→
   366→                <TabsContent value="TURBOPAY" className="mt-5 space-y-4">
   367→                  <div className="space-y-2">
   368→                    <Label htmlFor="tp-recipient">Recipient</Label>
   369→                    <div className="flex gap-2">
   370→                      <Input
   371→                        id="tp-recipient"
   372→                        placeholder="@username, phone, email or account no."
   373→                        value={tpRecipient}
   374→                        onChange={(e) => {
   375→                          setTpRecipient(e.target.value);
   376→                          setTpResolved(null);
   377→                        }}
   378→                        onKeyDown={(e) => {
   379→                          if (e.key === "Enter") resolveTurbopay();
   380→                        }}
   381→                      />
   382→                      <Button
   383→                        type="button"
   384→                        variant="secondary"
   385→                        onClick={resolveTurbopay}
   386→                        disabled={tpResolving || tpRecipient.trim().length < 3}
   387→                        className="gap-1.5 shrink-0"
   388→                      >
   389→                        {tpResolving ? (
   390→                          <RefreshCw className="h-4 w-4 animate-spin" />
   391→                        ) : (
   392→                          <Search className="h-4 w-4" />
   393→                        )}
   394→                        Resolve
   395→                      </Button>
   396→                    </div>
   397→                    {tpResolved && (
   398→                      <div className="flex items-center gap-2 rounded-lg border border-emerald-500/30 bg-emerald-500/10 px-3 py-2">
   399→                        <CheckCircle2 className="h-4 w-4 text-emerald-600 dark:text-emerald-400" />
   400→                        <span className="text-sm font-medium">{tpResolved.name}</span>
   401→                        <Badge variant="secondary" className="ml-auto">Turbopay user</Badge>
   402→                      </div>
   403→                    )}
   404→                  </div>
   405→                </TabsContent>
   406→
   407→                <TabsContent value="BANK" className="mt-5 space-y-4">
   408→                  <div className="space-y-2">
   409→                    <Label htmlFor="bank-select">Bank</Label>
   410→                    <Select value={bankCode} onValueChange={(v) => { setBankCode(v); setBankResolved(null); }}>
   411→                      <SelectTrigger id="bank-select" className="w-full">
   412→                        <SelectValue placeholder="Select bank" />
   413→                      </SelectTrigger>
   414→                      <SelectContent className="max-h-72">
   415→                        {UNIQUE_BANKS.map((b) => (
   416→                          <SelectItem key={b.code + b.name} value={b.code}>
   417→                            {b.name}
   418→                          </SelectItem>
   419→                        ))}
   420→                      </SelectContent>
   421→                    </Select>
   422→                  </div>
   423→                  <div className="space-y-2">
   424→                    <Label htmlFor="bank-account">Account number</Label>
   425→                    <div className="flex gap-2">
   426→                      <Input
   427→                        id="bank-account"
   428→                        inputMode="numeric"
   429→                        placeholder="0123456789"
   430→                        value={bankAccount}
   431→                        onChange={(e) => {
   432→                          setBankAccount(e.target.value.replace(/[^\d]/g, ""));
   433→                          setBankResolved(null);
   434→                        }}
   435→                        onKeyDown={(e) => {
   436→                          if (e.key === "Enter") resolveBank();
   437→                        }}
   438→                      />
   439→                      <Button
   440→                        type="button"
   441→                        variant="secondary"
   442→                        onClick={resolveBank}
   443→                        disabled={bankResolving || !bankCode || bankAccount.length < 6}
   444→                        className="gap-1.5 shrink-0"
   445→                      >
   446→                        {bankResolving ? (
   447→                          <RefreshCw className="h-4 w-4 animate-spin" />
   448→                        ) : (
   449→                          <Search className="h-4 w-4" />
   450→                        )}
   451→                        Resolve
   452→                      </Button>
   453→                    </div>
   454→                    {bankResolved && (
   455→                      <div className="flex items-center gap-2 rounded-lg border border-emerald-500/30 bg-emerald-500/10 px-3 py-2">
   456→                        <CheckCircle2 className="h-4 w-4 text-emerald-600 dark:text-emerald-400" />
   457→                        <span className="text-sm font-medium">{bankResolved.name}</span>
   458→                        <Badge variant="secondary" className="ml-auto">
   459→                          {bankResolved.bankName ?? "Bank"}
   460→                        </Badge>
   461→                      </div>
   462→                    )}
   463→                  </div>
   464→                </TabsContent>
   465→              </Tabs>
   466→
   467→              {/* Shared fields */}
   468→              <div className="mt-5 space-y-4">
   469→                <div className="space-y-2">
   470→                  <Label htmlFor="amount">Amount (₦)</Label>
   471→                  <Input
   472→                    id="amount"
   473→                    inputMode="numeric"
   474→                    placeholder="0.00"
   475→                    value={amountInput}
   476→                    onChange={(e) => setAmountInput(e.target.value)}
   477→                  />
   478→                  <div className="flex flex-wrap gap-1.5">
   479→                    {[1000, 5000, 10000, 25000, 50000].map((v) => (
   480→                      <button
   481→                        key={v}
   482→                        type="button"
   483→                        onClick={() => setAmountInput(String(v))}
   484→                        className="rounded-full border border-border bg-background px-2.5 py-1 text-xs font-medium hover:border-primary hover:bg-primary/5"
   485→                      >
   486→                        ₦{v.toLocaleString()}
   487→                      </button>
   488→                    ))}
   489→                  </div>
   490→                </div>
   491→
   492→                <div className="space-y-2">
   493→                  <Label htmlFor="note">Note (optional)</Label>
   494→                  <Input
   495→                    id="note"
   496→                    placeholder="What's this for?"
   497→                    value={note}
   498→                    onChange={(e) => setNote(e.target.value)}
   499→                    maxLength={100}
   500→                  />
   501→                </div>
   502→
   503→                {/* Fee + total */}
   504→                <div className="rounded-xl border bg-muted/40 p-3 text-sm">
   505→                  <div className="flex items-center justify-between">
   506→                    <span className="text-muted-foreground">Transfer fee</span>
   507→                    <span className="font-medium tabular-nums">
   508→                      {feeKobo > 0 ? naira(feeKobo) : "Free"}
   509→                    </span>
   510→                  </div>
   511→                  <div className="mt-1.5 flex items-center justify-between border-t pt-1.5">
   512→                    <span className="font-medium">Total debit</span>
   513→                    <span className="font-bold tabular-nums">{naira(totalKobo)}</span>
   514→                  </div>
   515→                </div>
   516→
   517→                <label className="flex items-center gap-2 text-sm">
   518→                  <Checkbox
   519→                    checked={saveBeneficiary}
   520→                    onCheckedChange={(v) => setSaveBeneficiary(v === true)}
   521→                  />
   522→                  Save as beneficiary
   523→                </label>
   524→
   525→                <Button
   526→                  className="w-full gap-1.5"
   527→                  size="lg"
   528→                  disabled={!canContinue() || submitting}
   529→                  onClick={onContinue}
   530→                >
   531→                  Continue <ArrowRight className="h-4 w-4" />
   532→                </Button>
   533→              </div>
   534→            </Card>
   535→          </div>
   536→
   537→          {/* Right column - beneficiaries */}
   538→          <div className="space-y-4">
   539→            <Card className="p-5">
   540→              <div className="mb-3 flex items-center justify-between">
   541→                <p className="text-sm font-semibold">Beneficiaries</p>
   542→                <Button
   543→                  size="sm"
   544→                  variant="ghost"
   545→                  className="h-7 gap-1.5 px-2"
   546→                  onClick={() => setView("beneficiaries")}
   547→                >
   548→                  <UserPlus className="h-3.5 w-3.5" /> Manage
   549→                </Button>
   550→              </div>
   551→              {benLoading ? (
   552→                <div className="space-y-2">
   553→                  {[0, 1, 2].map((i) => (
   554→                    <Skeleton key={i} className="h-14 w-full rounded-xl" />
   555→                  ))}
   556→                </div>
   557→              ) : beneficiaries.length === 0 ? (
   558→                <EmptyState
   559→                  icon={UserPlus}
   560→                  title="No beneficiaries yet"
   561→                  description="Saved recipients will appear here."
   562→                  action={
   563→                    <Button
   564→                      size="sm"
   565→                      variant="outline"
   566→                      onClick={() => setView("beneficiaries")}
   567→                    >
   568→                      Add beneficiary
   569→                    </Button>
   570→                  }
   571→                />
   572→              ) : (
   573→                <div className="max-h-96 space-y-1.5 overflow-y-auto scrollbar-thin pr-1">
   574→                  {beneficiaries.map((b) => (
   575→                    <div
   576→                      key={b.id}
   577→                      className="group flex items-center gap-3 rounded-xl border border-transparent p-2 transition-colors hover:border-border hover:bg-muted/40"
   578→                    >
   579→                      <button
   580→                        onClick={() => prefill(b)}
   581→                        className="flex min-w-0 flex-1 items-center gap-3 text-left"
   582→                      >
   583→                        <div className="flex h-9 w-9 shrink-0 items-center justify-center rounded-full bg-primary/10 text-primary">
   584→                          <span className="text-xs font-bold">
   585→                            {b.name.slice(0, 2).toUpperCase()}
   586→                          </span>
   587→                        </div>
   588→                        <div className="min-w-0 flex-1">
   589→                          <p className="truncate text-sm font-medium">{b.name}</p>
   590→                          <p className="truncate text-xs text-muted-foreground">
   591→                            {b.accountNumber} · {b.bankName}
   592→                          </p>
   593→                        </div>
   594→                      </button>
   595→                      <button
   596→                        onClick={() => toggleFavorite(b)}
   597→                        className="shrink-0 p-1.5"
   598→                        title={b.isFavorite ? "Unfavorite" : "Favorite"}
   599→                      >
   600→                        <Star
   601→                          className={`h-4 w-4 ${
   602→                            b.isFavorite
   603→                              ? "fill-amber-400 text-amber-400"
   604→                              : "text-muted-foreground"
   605→                          }`}
   606→                        />
   607→                      </button>
   608→                      <button
   609→                        onClick={() => deleteBeneficiary(b)}
   610→                        className="shrink-0 p-1.5 text-muted-foreground hover:text-destructive"
   611→                        title="Delete"
   612→                      >
   613→                        <Trash2 className="h-4 w-4" />
   614→                      </button>
   615→                    </div>
   616→                  ))}
   617→                </div>
   618→              )}
   619→            </Card>
   620→
   621→            <Card className="border-emerald-500/30 bg-emerald-500/5 p-5">
   622→              <div className="flex items-start gap-3">
   623→                <ShieldCheck className="h-5 w-5 text-emerald-600 dark:text-emerald-400" />
   624→                <div>
   625→                  <p className="text-sm font-semibold">Safe &amp; secure</p>
   626→                  <p className="mt-0.5 text-xs text-muted-foreground">
   627→                    All transfers require your 4-digit PIN. Bank transfers are
   628→                    protected by name verification.
   629→                  </p>
   630→                </div>
   631→              </div>
   632→            </Card>
   633→          </div>
   634→        </div>
   635→      )}
   636→
   637→      {/* Confirm dialog */}
   638→      <Dialog open={confirmOpen} onOpenChange={setConfirmOpen}>
   639→        <DialogContent className="max-w-sm">
   640→          <DialogHeader>
   641→            <DialogTitle>Confirm transfer</DialogTitle>
   642→            <DialogDescription>Review the details before continuing.</DialogDescription>
   643→          </DialogHeader>
   644→          <div className="space-y-3 py-2">
   645→            <Row label="Recipient" value={resolved?.name ?? recipientLabel} />
   646→            <Row
   647→              label="Account"
   648→              value={
   649→                type === "TURBOPAY"
   650→                  ? tpResolved?.username ?? tpRecipient
   651→                  : `${bankAccount} · ${bankResolved?.bankName ?? ""}`
   652→              }
   653→            />
   654→            <Row label="Amount" value={naira(amountKobo)} />
   655→            <Row label="Fee" value={feeKobo > 0 ? naira(feeKobo) : "Free"} />
   656→            {note && <Row label="Note" value={note} />}
   657→            <div className="rounded-xl border bg-muted/50 p-3">
   658→              <div className="flex items-center justify-between text-sm">
   659→                <span className="font-medium">Total debit</span>
   660→                <span className="text-lg font-bold tabular-nums">{naira(totalKobo)}</span>
   661→              </div>
   662→            </div>
   663→          </div>
   664→          <DialogFooter className="flex-col gap-2 sm:flex-row">
   665→            <Button variant="outline" className="flex-1" onClick={() => setConfirmOpen(false)}>
   666→              Cancel
   667→            </Button>
   668→            <Button
   669→              className="flex-1 gap-1.5"
   670→              disabled={submitting}
   671→              onClick={submitTransfer}
   672→            >
   673→              {submitting ? (
   674→                <RefreshCw className="h-4 w-4 animate-spin" />
   675→              ) : (
   676→                <ShieldCheck className="h-4 w-4" />
   677→              )}
   678→              Enter PIN
   679→            </Button>
   680→          </DialogFooter>
   681→        </DialogContent>
   682→      </Dialog>
   683→    </div>
   684→  );
   685→}
