   197→type AppNotification = {
   198→  id: string;
   199→  type: string; // TRANSACTION | SECURITY | KYC | REWARD | SYSTEM
   200→  title: string;

   208→// Resolve an actionUrl like "/history?ref=TP-XXX" into a setView key, or null.
   209→function resolveActionView(actionUrl: string | null): ViewKey | null {
   210→  if (!actionUrl) return null;
   211→  const clean = actionUrl.startsWith("/") ? actionUrl.slice(1) : actionUrl;
   212→  const seg = clean.split(/[/?#]/)[0];
   213→  if (seg && VALID_VIEW_KEYS.has(seg)) return seg as ViewKey;
   214→  return null;
   215→}

   217→// Notification icon + tone by type.
   218→function notifVisual(type: string): { Icon: any; tone: string } {
   219→  switch (type) {
   220→    case "TRANSACTION":
   221→      return { Icon: ArrowUpRight, tone: "emerald" };
   222→    case "SECURITY":
   223→      return { Icon: ShieldAlert, tone: "red" };
   224→    case "KYC":
   225→      return { Icon: BadgeCheck, tone: "amber" };
   226→    case "REWARD":
   227→      return { Icon: Gift, tone: "emerald" };
   228→    case "SYSTEM":
   229→    default:
   230→      return { Icon: Info, tone: "slate" };
   231→  }
   232→}

   313→        await fetch(`/api/notifications/${n.id}/read`, { method: "PATCH" });

   593→function SessionTimeoutDialog({
   594→  open,
   595→  secondsLeft,
   596→  onStay,
   597→  onSignOut,
   598→}: {
   599→  open: boolean;
   600→  secondsLeft: number;
   601→  onStay: () => void;
   602→  onSignOut: () => void;
   603→}) {
   604→  const minutes = Math.floor(secondsLeft / 60);
   605→  const seconds = secondsLeft % 60;
   606→  const timeLabel = `${minutes}:${seconds.toString().padStart(2, "0")}`;
   607→
   608→  // Countdown ring (SVG) — fraction of the 2-minute window remaining.
   609→  const totalSeconds = 120;
   610→  const fraction = Math.max(0, Math.min(1, secondsLeft / totalSeconds));
   611→  const radius = 36;
   612→  const circumference = 2 * Math.PI * radius;
   613→  const dashOffset = circumference * (1 - fraction);
   614→
   615→  // Amber/red transition when < 30s left
   616→  const urgent = secondsLeft <= 30;
   617→
   618→  return (
   619→    <Dialog open={open}>
   620→      <DialogContent className="max-w-sm gap-0 p-0" onPointerDownOutside={(e) => e.preventDefault()}>
   621→        {/* Amber header band */}
   622→        <div
   623→          className={`flex flex-col items-center gap-3 rounded-t-lg px-6 pb-5 pt-7 text-center ${
   624→            urgent
   625→              ? "bg-gradient-to-b from-red-500/15 to-transparent"
   626→              : "bg-gradient-to-b from-amber-500/15 to-transparent"
   627→          }`}
   628→        >
   629→          <div className="relative flex h-20 w-20 items-center justify-center">
   630→            <svg className="h-20 w-20 -rotate-90" viewBox="0 0 96 96">
   631→              <circle
   632→                cx="48"
   633→                cy="48"
   634→                r={radius}
   635→                fill="none"
   636→                stroke="currentColor"
   637→                strokeWidth="6"
   638→                className="text-muted/30"
   639→              />
   640→              <circle
   641→                cx="48"
   642→                cy="48"
   643→                r={radius}
   644→                fill="none"
   645→                stroke="currentColor"
   646→                strokeWidth="6"
   647→                strokeLinecap="round"
   648→                strokeDasharray={circumference}
   649→                strokeDashoffset={dashOffset}
   650→                className={urgent ? "text-red-500" : "text-amber-500"}
   651→                style={{ transition: "stroke-dashoffset 1s linear" }}
   652→              />
   653→            </svg>
   654→            <div className="absolute flex flex-col items-center">
   655→              {urgent ? (
   656→                <AlertTriangle className="h-5 w-5 text-red-500" />
   657→              ) : (
   658→                <Clock className="h-5 w-5 text-amber-500" />
   659→              )}
   660→              <span
   661→                className={`mt-0.5 font-mono text-base font-bold tabular-nums ${
   662→                  urgent ? "text-red-600 dark:text-red-400" : "text-amber-600 dark:text-amber-400"
   663→                }`}
   664→              >
   665→                {timeLabel}
   666→              </span>
   667→            </div>
   668→          </div>
   669→
   670→          <DialogHeader className="space-y-1.5 p-0">
   671→            <DialogTitle className="text-base">
   672→              Your session is about to expire
   673→            </DialogTitle>
   674→            <DialogDescription>
   675→              You&apos;ve been inactive for a while. For your security, we&apos;ll sign you out
   676→              automatically when the timer runs out.
   677→            </DialogDescription>
   678→          </DialogHeader>
   679→        </div>
   680→
   681→        <div className="space-y-2 px-6 py-4">
   682→          <Button onClick={onStay} className="w-full gap-1.5">
   683→            <ShieldCheck className="h-4 w-4" /> Stay signed in
   684→          </Button>
   685→          <Button
   686→            variant="outline"
   687→            onClick={onSignOut}
   688→            className="w-full gap-1.5 border-red-500/30 text-red-600 hover:bg-red-500/10 hover:text-red-700 dark:text-red-400"
   689→          >
   690→            <LogOut className="h-4 w-4" /> Sign out now
   691→          </Button>
   692→        </div>
   693→      </DialogContent>
   694→    </Dialog>
   695→  );
   696→}

   700→function NotificationCenterPanel({
   701→  open,
   702→  onOpenChange,
   703→  notifications,
   704→  loading,
   705→  unread,
   706→  filter,
   707→  onFilterChange,

   715→  loading: boolean;

   835→                      </div>

   855→    </Sheet>
