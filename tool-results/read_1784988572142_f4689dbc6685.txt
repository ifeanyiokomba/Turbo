   246→    }

   296→  }

   343→                <h2 className="text-sm font-semibold">Recent users</h2>

   370→                        </td>

   482→                        <th className="pb-2 pr-2 font-medium">Tier</th>
   483→                        <th className="pb-2 pr-2 font-medium">Balance</th>
   484→                        <th className="pb-2 pr-2 font-medium">Status</th>
   485→                        <th className="pb-2 font-medium">Joined</th>

   487→                    </thead>

   552→              <EmptyState icon={Users} title="No customers found" description="Try a different search." />
   553→            )}
   554→          </Card>
   555→        </TabsContent>
   556→
   557→        {/* Transactions */}
   558→        <TabsContent value="transactions" className="mt-5 space-y-4">
   559→          <Card className="p-4">
   560→            <div className="flex items-center gap-2">
   561→              <span className="text-xs text-muted-foreground">Filter:</span>
   562→              <Select value={txType} onValueChange={(v) => { setTxType(v); setTxPage(1); loadTxns(1, v); }}>
   563→                <SelectTrigger className="h-8 w-40">
   564→                  <SelectValue />
   565→                </SelectTrigger>
   566→                <SelectContent>
   567→                  <SelectItem value="ALL">All types</SelectItem>
   568→                  {Object.entries(TX_TYPE_LABELS).map(([v, l]) => (
   569→                    <SelectItem key={v} value={v}>{l}</SelectItem>
   570→                  ))}
   571→                </SelectContent>
   572→              </Select>
   573→            </div>
   574→          </Card>
   575→          <Card className="p-5">
   576→            {loadingTx && !txns ? (
   577→              <div className="space-y-2">
   578→                {[0,1,2,3,4].map((i) => <Skeleton key={i} className="h-12 rounded-xl" />)}
   579→              </div>
   580→            ) : txns && txns.transactions.length > 0 ? (
   581→              <>
   582→                <div className="overflow-x-auto">
   583→                  <table className="w-full text-sm">
   584→                    <thead>
   585→                      <tr className="text-left text-xs text-muted-foreground">
   586→                        <th className="pb-2 pr-2 font-medium">Reference</th>
   587→                        <th className="pb-2 pr-2 font-medium">User</th>
   588→                        <th className="pb-2 pr-2 font-medium">Type</th>
   589→                        <th className="pb-2 pr-2 font-medium">Amount</th>
   590→                        <th className="pb-2 pr-2 font-medium">Status</th>
   591→                        <th className="pb-2 font-medium">Date</th>
   592→                      </tr>
   593→                    </thead>
   594→                    <tbody>
   595→                      {txns.transactions.map((t) => (
   596→                        <tr key={t.id} className="border-t">
   597→                          <td className="py-2 pr-2">
   598→                            <p className="truncate font-mono text-xs">{t.reference}</p>
   599→                          </td>
   600→                          <td className="py-2 pr-2 text-xs">
   601→                            {t.userName ?? "—"}
   602→                            {t.userUsername && <span className="text-muted-foreground"> · @{t.userUsername}</span>}
   603→                          </td>
   604→                          <td className="py-2 pr-2 text-xs">{TX_TYPE_LABELS[t.type] ?? t.type}</td>
   605→                          <td className="py-2 pr-2 text-xs tabular-nums">
   606→                            {naira(t.amountKobo)}
   607→                            {t.feeKobo > 0 && <span className="text-muted-foreground"> +{nairaCompact(t.feeKobo)} fee</span>}
   608→                          </td>
   609→                          <td className="py-2 pr-2">
   610→                            <Badge variant="secondary" className={`text-[10px] ${STATUS_TONE[t.status] ?? ""}`}>
   611→                              {t.status}
   612→                            </Badge>
   613→                          </td>
   614→                          <td className="py-2 text-xs text-muted-foreground">{formatDate(t.createdAt, true)}</td>
   615→                        </tr>
   616→                      ))}
   617→                    </tbody>
   618→                  </table>
   619→                </div>
   620→                <div className="mt-4 flex items-center justify-between text-xs text-muted-foreground">
   621→                  <p>{txns.total} total · page {txns.page}</p>
   622→                  <div className="flex gap-2">
   623→                    <Button
   624→                      size="sm"
   625→                      variant="outline"
   626→                      disabled={txns.page <= 1 || loadingTx}
   627→                      onClick={() => {
   628→                        const p = Math.max(1, txns.page - 1);
   629→                        setTxPage(p);
   630→                        loadTxns(p, txType);
   631→                      }}
   632→                    >
   633→                      Previous
   634→                    </Button>
   635→                    <Button
   636→                      size="sm"
   637→                      variant="outline"
   638→                      disabled={!txns.hasMore || loadingTx}
   639→                      onClick={() => {
   640→                        const p = txns.page + 1;
   641→                        setTxPage(p);
   642→                        loadTxns(p, txType);
   643→                      }}
   644→                    >
   645→                      Next
   646→                    </Button>
   647→                  </div>
   648→                </div>
   649→              </>
   650→            ) : (
   651→              <EmptyState icon={ArrowLeftRight} title="No transactions found" description="Try a different filter." />
   652→            )}
   653→          </Card>
   654→        </TabsContent>
   655→
   656→        {/* AML */}
   657→        <TabsContent value="aml" className="mt-5">
   658→          <Card className="p-5">
   659→            <div className="mb-4 flex items-center gap-2">
   660→              <ShieldAlert className="h-5 w-5 text-primary" />
   661→              <h2 className="text-base font-semibold">AML flags feed</h2>
   662→              <Badge variant="secondary" className="ml-auto">
   663→                {data?.stats.amlFlags ?? 0} unresolved
   664→              </Badge>
   665→            </div>
   666→            {data?.amlFlags && data.amlFlags.length > 0 ? (
   667→              <ul className="space-y-2">
   668→                {data.amlFlags.map((f) => (
   669→                  <li key={f.id} className="flex items-center gap-3 rounded-xl border p-3">
   670→                    <div className={`flex h-10 w-10 shrink-0 items-center justify-center rounded-full ${SEVERITY_TONE[f.severity] ?? "bg-muted text-muted-foreground"}`}>
   671→                      <AlertTriangle className="h-5 w-5" />
   672→                    </div>
   673→                    <div className="min-w-0 flex-1">
   674→                      <p className="truncate text-sm font-medium">
   675→                        {f.rule.replace(/_/g, " ").toLowerCase().replace(/\b\w/g, (c) => c.toUpperCase())}
   676→                      </p>
   677→                      <p className="truncate text-xs text-muted-foreground">
   678→                        {f.userName ?? "Unknown user"}{f.userUsername ? ` · @${f.userUsername}` : ""} · {f.description}
   679→                      </p>
   680→                      <p className="mt-0.5 text-xs text-muted-foreground">{timeAgo(f.createdAt)} · {formatDate(f.createdAt, true)}</p>
   681→                    </div>
   682→                    <Badge variant="outline" className={`text-[10px] ${SEVERITY_TONE[f.severity] ?? ""}`}>
   683→                      {f.severity}
   684→                    </Badge>
   685→                  </li>
   686→                ))}
   687→              </ul>
   688→            ) : (
   689→              <EmptyState
   690→                icon={ShieldAlert}
   691→                title="No unresolved AML flags"
   692→                description="All clear — no suspicious activity pending review."
   693→              />
   694→            )}
   695→          </Card>
   696→        </TabsContent>
   697→      </Tabs>
   698→    </div>
   699→  );
