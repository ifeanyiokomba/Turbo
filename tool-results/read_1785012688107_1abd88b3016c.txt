   245→// Tiny CSV export helper — escapes commas/quotes/newlines and triggers a download.
   246→function exportCsv(
   247→  filename: string,
   248→  headers: string[],
   249→  rows: (string | number | null | undefined)[][],
   250→) {
   251→  const escape = (v: string | number | null | undefined) => {
   252→    const s = v == null ? "" : String(v);
   253→    if (/[",\n]/.test(s)) return `"${s.replace(/"/g, '""')}"`;
   254→    return s;
   255→  };
   256→  const lines = [
   257→    headers.map(escape).join(","),
   258→    ...rows.map((r) => r.map(escape).join(",")),
   259→  ];
   260→  const blob = new Blob([lines.join("\n")], {
   261→    type: "text/csv;charset=utf-8;",
   262→  });
   263→  const url = URL.createObjectURL(blob);
   264→  const a = document.createElement("a");
   265→  a.href = url;
   266→  a.download = filename;
   267→  document.body.appendChild(a);
   268→  a.click();
   269→  document.body.removeChild(a);
   270→  URL.revokeObjectURL(url);
   271→}
